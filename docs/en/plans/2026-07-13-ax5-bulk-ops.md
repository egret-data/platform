- [ ] **Step 3: Create the `runBulk` helper**

`frontend/src/lib/runBulk.ts`:
```tsx
import { toast } from "sonner";

export interface BulkResult<T> {
  ok: T[];
  failed: T[];
}

/**
 * Run `op` over each item independently (collecting successes + failures), refresh,
 * then toast a summary. When `undo` is provided and at least one item succeeded, the
 * toast offers an Undo that runs `undo` over only the SUCCEEDED items.
 */
export async function runBulk<T>(opts: {
  items: T[];
  op: (item: T) => Promise<unknown>;
  verb: string; // e.g. "Disabled"
  noun: string; // e.g. "users"
  refresh: () => void;
  undo?: (item: T) => Promise<unknown>;
}): Promise<BulkResult<T>> {
  const ok: T[] = [];
  const failed: T[] = [];
  for (const item of opts.items) {
    try {
      await opts.op(item);
      ok.push(item);
    } catch {
      failed.push(item);
    }
  }
  opts.refresh();

  const text =
    failed.length === 0
      ? `${opts.verb} ${ok.length} ${opts.noun}.`
      : `${opts.verb} ${ok.length} of ${opts.items.length} ${opts.noun}; ${failed.length} failed.`;

  const action =
    opts.undo && ok.length > 0
      ? {
          label: "Undo",
          onClick: async () => {
            for (const item of ok) {
              try {
                await opts.undo!(item);
              } catch {
                /* best-effort undo */
              }
            }
            opts.refresh();
            toast.success("Undone");
          },
        }
      : undefined;

  if (failed.length) toast.error(text, action ? { action } : undefined);
  else toast.success(text, action ? { action } : undefined);

  return { ok, failed };
}
```

`frontend/src/lib/__tests__/runBulk.test.ts` (mock sonner):
```tsx
import { runBulk } from "@/lib/runBulk";
vi.mock("sonner", () => ({ toast: Object.assign(vi.fn(), { success: vi.fn(), error: vi.fn() }) }));
import { toast } from "sonner";

it("runs op per item, refreshes, and reports success", async () => {
  const op = vi.fn().mockResolvedValue(undefined);
  const refresh = vi.fn();
  const res = await runBulk({ items: ["a", "b"], op, verb: "Disabled", noun: "users", refresh });
  expect(op).toHaveBeenCalledTimes(2);
  expect(refresh).toHaveBeenCalled();
  expect(res.ok).toEqual(["a", "b"]);
  expect(toast.success).toHaveBeenCalledWith("Disabled 2 users.", undefined);
});

it("reports partial failure and offers Undo over succeeded items only", async () => {
  const op = vi.fn().mockImplementation((x: string) => (x === "b" ? Promise.reject(new Error()) : Promise.resolve()));
  const undo = vi.fn().mockResolvedValue(undefined);
  const refresh = vi.fn();
  const res = await runBulk({ items: ["a", "b"], op, verb: "Disabled", noun: "users", refresh, undo });
  expect(res.ok).toEqual(["a"]);
  expect(res.failed).toEqual(["b"]);
  const [msg, cfg] = (toast.error as unknown as { mock: { calls: unknown[][] } }).mock.calls.at(-1)!;
  expect(msg).toBe("Disabled 1 of 2 users; 1 failed.");
  // Undo action inverts only the succeeded item ("a").
  await (cfg as { action: { onClick: () => Promise<void> } }).action.onClick();
  expect(undo).toHaveBeenCalledTimes(1);
  expect(undo).toHaveBeenCalledWith("a");
});
```
