**Goal:** Add a Map (choropleth) visualization with three selectable maps — US states, World countries, Vietnam provinces — shading regions by a metric.

**Architecture:** Reuse ECharts (`map` series + `VisualMap`). A committed prep script generates three geojson assets keyed by the exact code/name the data uses (`properties.key`). **Tech Stack:** React 18 + TS + Vite + Vitest + ECharts 6 (already a dependency) + lucide-react. Python 3 for the one-time asset prep script.

```python
#!/usr/bin/env python3
"""Download + normalize the map geojson assets for the Map viz (run once; outputs are committed).

Each feature's properties are reduced to {key, name}:
  - US states:        key = 2-letter state abbreviation, name = full state name
  - World countries:  key = ISO-A2 (ISO_A2_EH fixes Natural Earth's -99 gaps), name = country name
  - Vietnam provinces: key = province name (shapeName), name = province name
Coordinates are rounded to 2 decimals to shrink the files. Requires network.
"""
import json
import os
import urllib.request

OUT = os.path.join(os.path.dirname(__file__), "..", "frontend", "src", "components", "chart", "echarts", "maps")

US_ABBR = {
    "Alabama": "AL", "Alaska": "AK", "Arizona": "AZ", "Arkansas": "AR", "California": "CA",
    "Colorado": "CO", "Connecticut": "CT", "Delaware": "DE", "District of Columbia": "DC",
    "Florida": "FL", "Georgia": "GA", "Hawaii": "HI", "Idaho": "ID", "Illinois": "IL",
    "Indiana": "IN", "Iowa": "IA", "Kansas": "KS", "Kentucky": "KY", "Louisiana": "LA",
    "Maine": "ME", "Maryland": "MD", "Massachusetts": "MA", "Michigan": "MI", "Minnesota": "MN",
    "Mississippi": "MS", "Missouri": "MO", "Montana": "MT", "Nebraska": "NE", "Nevada": "NV",
    "New Hampshire": "NH", "New Jersey": "NJ", "New Mexico": "NM", "New York": "NY",
    "North Carolina": "NC", "North Dakota": "ND", "Ohio": "OH", "Oklahoma": "OK", "Oregon": "OR",
    "Pennsylvania": "PA", "Rhode Island": "RI", "South Carolina": "SC", "South Dakota": "SD",
    "Tennessee": "TN", "Texas": "TX", "Utah": "UT", "Vermont": "VT", "Virginia": "VA",
    "Washington": "WA", "West Virginia": "WV", "Wisconsin": "WI", "Wyoming": "WY", "Puerto Rico": "PR",
}


def fetch(url):
    with urllib.request.urlopen(url, timeout=60) as r:
        return json.loads(r.read().decode())


def round_coords(obj, nd=2):
    if isinstance(obj, list):
        if obj and isinstance(obj[0], (int, float)):
            return [round(float(x), nd) for x in obj]
        return [round_coords(x, nd) for x in obj]
    return obj


def normalize(features, key_of, name_of):
    out = []
    for f in features:
        p = f.get("properties", {})
        key = key_of(p)
        name = name_of(p)
        if not key:
            continue
        out.append({
            "type": "Feature",
            "properties": {"key": str(key), "name": str(name)},
            "geometry": {**f["geometry"], "coordinates": round_coords(f["geometry"]["coordinates"])},
        })
    return {"type": "FeatureCollection", "features": out}


def write(name, fc):
    os.makedirs(OUT, exist_ok=True)
    path = os.path.join(OUT, name)
    with open(path, "w") as fh:
        json.dump(fc, fh, separators=(",", ":"))
    print(f"  wrote {name}: {len(fc['features'])} features, {os.path.getsize(path)//1024} KB")


def main():
    us = fetch("https://raw.githubusercontent.com/PublicaMundi/MappingAPI/master/data/geojson/us-states.json")
    write("us-states.geo.json", normalize(
        us["features"],
        key_of=lambda p: US_ABBR.get(p.get("name", "")),
        name_of=lambda p: p.get("name", ""),
    ))

    world = fetch("https://raw.githubusercontent.com/nvkelso/natural-earth-vector/master/geojson/ne_110m_admin_0_countries.geojson")
    write("world-countries.geo.json", normalize(
        world["features"],
        key_of=lambda p: (p.get("ISO_A2_EH") or p.get("ISO_A2") or "").upper() if (p.get("ISO_A2_EH") or p.get("ISO_A2") or "") not in ("-99", "") else None,
        name_of=lambda p: p.get("NAME", ""),
    ))

    vmeta = fetch("https://www.geoboundaries.org/api/current/gbOpen/VNM/ADM1/")
    vurl = vmeta["gjDownloadURL"] if isinstance(vmeta, dict) else vmeta[0]["gjDownloadURL"]
    vn = fetch(vurl)
    write("vietnam-provinces.geo.json", normalize(
        vn["features"],
        key_of=lambda p: p.get("shapeName", ""),
        name_of=lambda p: p.get("shapeName", ""),
    ))


if __name__ == "__main__":
    main()
```

```ts
import type { EChartsCoreOption } from "echarts/core";
import type { ResultColumn } from "@/components/chart/ChartTable";

export interface MapBuildConfig {
  echartsName: string;
  region?: string;
  metric?: string;
  normalize: (v: unknown) => string;
}

/** Build a choropleth `map` option. The series data names are normalized region keys that match
 *  the registered geojson's `properties.key`; unmatched/NaN rows are dropped. */
export function buildMapOption(columns: ResultColumn[], rows: unknown[][], cfg: MapBuildConfig): EChartsCoreOption {
  const idx = (name?: string) => columns.findIndex((c) => c.name === name);
  const ri = idx(cfg.region);
  const mi = idx(cfg.metric);
  const data =
    ri < 0 || mi < 0
      ? []
      : rows
          .map((r) => ({ name: cfg.normalize(r[ri]), value: Number(r[mi]) }))
          .filter((d) => d.name !== "" && Number.isFinite(d.value));
  const values = data.map((d) => d.value);
  const min = values.length ? Math.min(...values) : 0;
  const max = values.length ? Math.max(...values) : 1;
  return {
    tooltip: {
      trigger: "item",
      formatter: (p: { name?: string; value?: number; data?: { name: string } }) =>
        p.value == null || Number.isNaN(p.value) ? String(p.name ?? "") : `${p.name}: ${Number(p.value).toLocaleString()}`,
    },
    visualMap: {
      min,
      max,
      left: 8,
      bottom: 8,
      calculable: true,
      inRange: { color: ["#e6f4f1", "#0f9b8e"] },
      text: ["High", "Low"],
    },
    series: [
      {
        type: "map",
        map: cfg.echartsName,
        nameProperty: "key",
        roam: false,
        emphasis: { label: { show: false } },
        data,
      },
    ],
  };
}
```

`frontend/src/components/chart/MapChart.tsx`:

```tsx
import { useEffect, useState } from "react";
import { registerMap } from "echarts/core";
import type { ChartParams } from "@/api/charts";
import { EChartsRenderer } from "@/components/chart/echarts/EChartsRenderer";
import { buildMapOption } from "@/components/chart/echarts/options/map";
import { MAPS } from "@/components/chart/echarts/maps/registry";
import type { ResultColumn } from "@/components/chart/ChartTable";

const registered = new Set<string>();

/** Choropleth map. Lazy-loads + registers the selected geojson, then renders via EChartsRenderer. */
export function MapChart({ columns, rows, params }: { columns: ResultColumn[]; rows: unknown[][]; params: ChartParams }) {
  const def = MAPS[params.map?.mapKey ?? "us_states"] ?? MAPS.us_states;
  const [ready, setReady] = useState(registered.has(def.echartsName));
  const [error, setError] = useState(false);

  useEffect(() => {
    let alive = true;
    if (registered.has(def.echartsName)) {
      setReady(true);
      return;
    }
    setReady(false);
    setError(false);
    def
      .load()
      .then((geojson) => {
        if (!alive) return;
        registerMap(def.echartsName, geojson as never);
        registered.add(def.echartsName);
        setReady(true);
      })
      .catch(() => alive && setError(true));
    return () => {
      alive = false;
    };
  }, [def]);

  if (error) return <div className="flex h-full items-center justify-center text-sm text-destructive">Couldn't load the map</div>;
  if (!ready) return <div className="flex h-full items-center justify-center text-sm text-muted-foreground">Loading map…</div>;
  if (rows.length === 0) return <div className="flex h-full items-center justify-center text-sm text-muted-foreground">No data</div>;

  return (
    <div className="h-full w-full">
      <EChartsRenderer
        option={buildMapOption(columns, rows, {
          echartsName: def.echartsName,
          region: params.map?.region,
          metric: params.map?.metric,
          normalize: def.normalize,
        })}
      />
    </div>
  );
}
```
